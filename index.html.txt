function taskHTML(x){

function renderWorker(){

  if(!m){box.innerHTML='<div class="empty">'+t('pickMeMsg')+'</div>';return}
  var list=tasks.filter(function(x){return x.workerId===m&&(x.date===TODAY||(x.date<TODAY&&x.status!=='done'))}).sort(function(a,b){return (a.date>b.date)-(a.date<b.date)||(a.createdAt||0)-(b.createdAt||0)});
  if(!list.length){box.innerHTML='<div class="empty">'+t('noMine')+'</div>';return}
  box.innerHTML=list.map(function(x){
    var s=x.status||'todo',a='',b='';
    if(x.date<TODAY)a='<div class="note p" style="margin-bottom:8px">'+t('carry')+' '+esc(x.date)+'</div>';
    if(s==='todo'||s==='problem')b='<button class="go" data-st="doing" data-id="'+x.id+'">'+t('start')+'</button>';
    if(s==='doing')b='<button class="ok" data-st="done" data-id="'+x.id+'">'+t('finish')+'</button>';
    if(s!=='done')b+='<button class="bad" data-st="problem" data-id="'+x.id+'">'+t('prob')+'</button>';
    else b='<button data-st="doing" data-id="'+x.id+'">'+t('undo')+'</button>';

  }).join('');
}
